import streamlit as st
from chatbot_logic import RuleBasedChatbot

# Configure the Streamlit page
st.set_page_config(page_title="Rule-Based Chatbot", page_icon="🤖", layout="centered")

st.title("🤖 Rule-Based Chatbot")
st.write("Welcome! This chatbot uses Python regular expressions and predefined rules to answer your questions. Try asking about AI, saying hello, or asking for help.")

# Initialize the chatbot instance in session state so it persists
if "bot" not in st.session_state:
    st.session_state.bot = RuleBasedChatbot()

# Initialize chat history list in session state
if "messages" not in st.session_state:
    st.session_state.messages = []

# Display previous chat messages from history
for message in st.session_state.messages:
    with st.chat_message(message["role"]):
        st.markdown(message["content"])

# React to user input at the bottom of the screen
if user_input := st.chat_input("Type your message here..."):
    # 1. Display user message in chat container
    st.chat_message("user").markdown(user_input)
    # 2. Add user message to chat history memory
    st.session_state.messages.append({"role": "user", "content": user_input})

    # 3. Generate bot response using the rule-based logic
    response = st.session_state.bot.get_response(user_input)

    # 4. Display assistant response in chat container
    with st.chat_message("assistant"):
        st.markdown(response)
    # 5. Add assistant response to chat history memory
    st.session_state.messages.append({"role": "assistant", "content": response})
